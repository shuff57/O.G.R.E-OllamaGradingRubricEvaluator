### ogre-desktop/src/lib/browser-actions.ts
- Replaced the 4-line `navigateAction(url)` stub with a full implementation:
  - Signature: `navigateAction(url: string, timeout = 10_000): Promise<ActionResult>`
  - Calls `Page.enable` before registering any listener
  - Registers `Page.loadEventFired` listener BEFORE sending `Page.navigate`
    (race-condition safe — the load event cannot fire before the listener is in place)
  - Immediately checks `Page.navigate` response for `errorText` → returns
    `{ success: false, error }` without waiting, cleans up listener
  - Uses `Promise.race([loadPromise, timeoutPromise])` pattern
  - Timeout path: resolves `{ success: true, warning: 'Page load timed out after Xms' }`
  - Success path: resolves `{ success: true }`
  - Both paths: `cdp.off('Page.loadEventFired', onLoad)` + `clearTimeout` in every branch
    (no listener leaks)

## Implementation Pattern

```typescript
async function navigateAction(url: string, timeout = 10_000): Promise<ActionResult> {
  try {
    await cdp.send('Page.enable');

    let loadResolve: () => void;
    const loadPromise = new Promise<void>((res) => { loadResolve = res; });
    const onLoad = () => { loadResolve(); };

    cdp.on('Page.loadEventFired', onLoad);   // register BEFORE navigate

    let navResult: unknown;
    try {
      navResult = await cdp.send('Page.navigate', { url });
    } catch (err) {
      cdp.off('Page.loadEventFired', onLoad);
      return { success: false, error: ... };
    }

    if (navResult?.errorText) {
      cdp.off('Page.loadEventFired', onLoad);
      return { success: false, error: navResult.errorText };
    }

    let timeoutHandle: ReturnType<typeof setTimeout>;
    const timeoutPromise = new Promise<'timeout'>((res) => {
      timeoutHandle = setTimeout(() => res('timeout'), timeout);
    });

    const winner = await Promise.race([loadPromise.then(() => 'loaded'), timeoutPromise]);
    cdp.off('Page.loadEventFired', onLoad);
    clearTimeout(timeoutHandle);

    return winner === 'timeout'
      ? { success: true, warning: `Page load timed out after ${timeout}ms` }
      : { success: true };
  } catch (err) {
    return { success: false, error: ... };
  }
}
```
